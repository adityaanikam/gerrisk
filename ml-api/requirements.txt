# GeriRisk ML API — Python dependencies
#
# IMPORTANT: Requires Python 3.12 (see .python-version).
# These versions all ship pre-built binary wheels for Python 3.12 on PyPI,
# so no Rust/C compiler is needed during deployment on Render.
#
# Python 3.14 is NOT supported: pydantic-core ~=2.9 and numpy ~=1.26
# pre-date Python 3.14 and have no pre-built wheels for it, causing
# "Read-only file system" Cargo build failures on Render's free tier.

fastapi~=0.115.0
uvicorn[standard]~=0.32.0    # [standard] pulls in httptools + uvloop for speed
pydantic~=2.9.0

# Exact pin, not a range: every artefact in models/ carries
# _sklearn_version == "1.6.1", and loading a pickle under a different
# scikit-learn raises InconsistentVersionWarning with no guarantee that the
# unpickled estimator still behaves the same. Bump this only together with
# retrained artefacts (see train.py).
scikit-learn==1.6.1

numpy~=1.26.4
joblib~=1.4.2
